pybind11是一个较为好用的实现python调用C++接口的函数库，在Pytorch中使用。这里不仅仅适用于Pytorch。

<!--more-->

* 优点：轻量级，只需要包含头文件，不需要链接其他库。
* 支持很多C++数据结构和Python数据结构之间的原生转换。比如C++中的std::vector，array能够和Python中的list。std::map和Python的dict转换。
* 支持很多C++11的特性，比如智能指针，lambda函数等。



```c++
#include <pybind11/pubind11.h>
int add(int i, int j){
    return i+j;
}

PYBIND11_MODULE(example,m){
    m.doc() = "pybind11 example plugin";
    m.def("add", &add, "A function which adds two numbers");
}
```

```bash
g++ -O3 -Wall-shared-std=c++11 -fPIC 'python3 -m pubind11 --includes' example.cc -o example 'python3-config--extention-suffix'
```

```python
import example 
example.add()
```



使用pybind11导出C++的类：

```c++
#include <pybind11/pybind11.h>
#include <pybind11/stl.h>
#include <pybind11/operator.h>

namespace py = pybind11;

template<typename T>
class vec3 {
    public:
    	vec3(Tx, Ty, Tz): _x(x),_y(y),_z(z);
    	
    	vec3<T> operator+(const vec3<T> & o)const{ //加上const说明代码是只读的
            return vec3<T>(_x + o._x, _y+o._y, _z+o._z);
    	}
    
    std::string toString() const{
        return "vec3:[" +
            	std::to_String(_x) + ","
            	+std::to_String(_y) + ","
            	+std::to_String(_z) + "]";
    }
    
    std::tuple toTuple() const{
        return std::make_tuple(_x,_y,_z);
    }
    
    private:
    	T _x, _y, _z;
}

PYBIND11_MODULE(example, m){
    m.doc()="Example";
    py::class_<vec3<float>>(m,"Vec3")
        .def(py::init<float,float,float>())
        .def(py::self + py::self) 
        .def("to_tuple",&vec<float>::toTuple)// 调用stl.h，进行自动的转换。
        .def("__repr__",&vec<float>::toString);
}

```
